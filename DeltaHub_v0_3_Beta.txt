-- Effects are placed DIRECTLY in Lighting (not in a Folder),
-- because post-processing effects only work there.
local function clearAllEffects()
    for _, child in ipairs(Lighting:GetChildren()) do
        if child.Name:sub(1, #EFFECT_PREFIX) == EFFECT_PREFIX then
            child:Destroy()
        end
    end

    local legacy = Lighting:FindFirstChild("DeltaHubEffects")
    if legacy then
        legacy:Destroy()
    end
end

local function create(class, props, parent)
    local obj = Instance.new(class)

    for k, v in pairs(props or {}) do
        obj[k] = v
    end

    obj.Parent = parent
    return obj
end

local function round(obj, radius)
    create("UICorner", {
        CornerRadius = UDim.new(0, radius or 10)
    }, obj)
end

local function animate(obj, props, duration)
    TweenService:Create(
        obj,
        TweenInfo.new(
            duration or 0.2,
            Enum.EasingStyle.Quart,
            Enum.EasingDirection.Out
        ),
        props
    ):Play()
end

local function label(parent, text, height, color)
    return create("TextLabel", {
        Size = UDim2.new(1, 0, 0, height or 30),
        BackgroundTransparency = 1,
        Text = text,
        TextColor3 = color or C.Text,
        TextSize = 13,
        Font = Enum.Font.Gotham,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Top,
        TextWrapped = true
    }, parent)
end

-- isActive (optional): function returning true when the button
-- should keep the accent colour (toggles, selected tab, favourites).
local function button(parent, text, callback, isActive)
    local function baseColor()
        if isActive and isActive() then
            return C.Accent
        end
        return C.Button
    end

    local b = create("TextButton", {
        Size = UDim2.new(1, 0, 0, 38),
        BackgroundColor3 = baseColor(),
        Text = text,
        TextColor3 = C.Text,
        TextSize = 13,
        Font = Enum.Font.GothamMedium,
        AutoButtonColor = false
    }, parent)

    round(b, 9)

    b.MouseEnter:Connect(function()
        animate(b, {BackgroundColor3 = C.Accent})
    end)

    b.MouseLeave:Connect(function()
        animate(b, {BackgroundColor3 = baseColor()})
    end)

    b.Activated:Connect(function()
        if callback then
            callback()
        end
    end)

    return b
end

end

        math.min((v.X - 24) / 650, (v.Y - 24) / 430, 1),

        cameraConnection = nil

        cameraConnection = camera:GetPropertyChangedSignal("ViewportSize"):Connect(updateScale)

local EffectState = {}
local VisualButtons = {}

local function toggleEffect(name, enabled)
    local fullName = EFFECT_PREFIX .. name

    local oldEffect = Lighting:FindFirstChild(fullName)
    if oldEffect then
        oldEffect:Destroy()
    end

    if not enabled then return end

    if name == "Bloom" then
        create("BloomEffect", {
            Name = fullName,
            Intensity = 0.8,
            Size = 32,
            Threshold = 0.8
        }, Lighting)

    elseif name == "SunRays" then
        create("SunRaysEffect", {
            Name = fullName,
            Intensity = 0.25,
            Spread = 1
        }, Lighting)

    elseif name == "ColorCorrection" then
        create("ColorCorrectionEffect", {
            Name = fullName,
            Brightness = 0.03,
            Contrast = 0.15,
            Saturation = 0.2,
            TintColor = Color3.fromRGB(255, 244, 225)
        }, Lighting)

    elseif name == "Atmosphere" then
        create("Atmosphere", {
            Name = fullName,
            Density = 0.3,
            Offset = 0.1,
            Color = Color3.fromRGB(205, 220, 255),
            Decay = Color3.fromRGB(100, 110, 140),
            Haze = 1.2,
            Glare = 0.15
        }, Lighting)

    elseif name == "Brightness" then
        create("ColorCorrectionEffect", {
            Name = fullName,
            Brightness = 0.12
        }, Lighting)

    elseif name == "Contrast" then
        create("ColorCorrectionEffect", {
            Name = fullName,
            Contrast = 0.2
        }, Lighting)

    elseif name == "Saturation" then
        create("ColorCorrectionEffect", {
            Name = fullName,
            Saturation = 0.3
        }, Lighting)
    end
end

local function setVisual(name, enabled)
    EffectState[name] = enabled
    toggleEffect(name, enabled)

    local b = VisualButtons[name]
    if b then
        b.Text = name .. (enabled and "  [ON]" or "  [OFF]")
        animate(b, {BackgroundColor3 = enabled and C.Accent or C.Button})
    end
end

        button(
            FriendList,

                (Favorites[id] and "  ★" or ""),
            function()
                Favorites[id] = not Favorites[id]
                refreshFriends()
            end,
            function()
                return Favorites[id] == true
            end
        )
    end

button(FriendsPage, "Добавить игрока", function()
    local query = FriendInput.Text
    if query == "" then return end

    local found

    for _, p in ipairs(Players:GetPlayers()) do
        if p.Name:lower() == query:lower()
            or tostring(p.UserId) == query then
            found = p
            break
        end
    end

    if found then
        Friends[found.UserId] = {Name = found.Name}
        FriendInput.Text = ""
        refreshFriends()
    else
        warn("Игрок не найден в текущем сервере.")
    end
end)

local function requestAPI(method, path, body)
    if CONFIG.API_BASE:find("YOUR%-DELTAHUB") then
        return false, "API address is not configured"
    end

    local requestFunction =
        (syn and syn.request)
        or (http and http.request)
        or request

    if not requestFunction then
        return false, "HTTP request is unavailable"
    end

    local success, response = pcall(function()
        return requestFunction({
            Url = CONFIG.API_BASE .. path,
            Method = method,
            Headers = {
                ["Content-Type"] = "application/json",
                ["X-DeltaHub-UserId"] = tostring(Player.UserId)
            },
            Body = body and HttpService:JSONEncode(body) or nil
        })
    end)

    if not success then
        return false, tostring(response)
    end

    if not response or response.StatusCode < 200
        or response.StatusCode >= 300 then
        return false, "HTTP request failed"
    end

    local decodedOK, decoded = pcall(function()
        return HttpService:JSONDecode(response.Body)
    end)

    if not decodedOK then
        return false, "Invalid API response"
    end

    return true, decoded
end

local function refreshPremium()
    PremiumStatus.Text = "Проверяем Premium..."
    PremiumStatus.TextColor3 = C.Accent

    task.spawn(function()
        local ok, result = requestAPI(
            "GET",
            "/v1/premium/" .. tostring(Player.UserId)
        )

        if not ok then
            PremiumStatus.Text = "Premium: сервис не подключён"
            PremiumStatus.TextColor3 = C.Muted
            PremiumExpiry.Text = tostring(result)
            return
        end

        if result.active then
            PremiumStatus.Text = "Premium: АКТИВЕН"
            PremiumStatus.TextColor3 = C.Green

            PremiumExpiry.Text =
                result.expiresAt and
                ("Действует до: " .. tostring(result.expiresAt))
                or "Срок: навсегда"
        else
            PremiumStatus.Text = "Premium: НЕ АКТИВЕН"
            PremiumStatus.TextColor3 = C.Muted
            PremiumExpiry.Text = "Подписка отсутствует или истекла"
        end
    end)
end

-- DRAGGING (mouse and touch)

local dragging = false
local dragStart
local startPosition

track(Top.InputBegan:Connect(function(inputObject)
    if inputObject.UserInputType == Enum.UserInputType.MouseButton1
        or inputObject.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = inputObject.Position
        startPosition = Main.Position
    end
end))

track(UserInputService.InputChanged:Connect(function(inputObject)
    if dragging and (
        inputObject.UserInputType == Enum.UserInputType.MouseMovement
        or inputObject.UserInputType == Enum.UserInputType.Touch
    ) then
        local delta = inputObject.Position - dragStart

        Main.Position = UDim2.new(
            startPosition.X.Scale, startPosition.X.Offset + delta.X,
            startPosition.Y.Scale, startPosition.Y.Offset + delta.Y
        )
    end
end))

track(UserInputService.InputEnded:Connect(function(inputObject)
    if inputObject.UserInputType == Enum.UserInputType.MouseButton1
        or inputObject.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end))
