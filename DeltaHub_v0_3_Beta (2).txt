-- Effects are placed DIRECTLY in Lighting (not in a Folder),
-- because post-processing effects only work there.
local function clearAllEffects()
    for _, child in ipairs(Lighting:GetChildren()) do
        if child.Name:sub(1, #EFFECT_PREFIX) == EFFECT_PREFIX then
            child:Destroy()
        end
    end

    local legacy = Lighting:FindFirstChild("DeltaHubEffects")
    if legacy then
        legacy:Destroy()
    end
end

clearAllEffects()

local function create(class, props, parent)
    local obj = Instance.new(class)

    for k, v in pairs(props or {}) do
        obj[k] = v
    end

    obj.Parent = parent
    return obj
end

local function round(obj, radius)
    create("UICorner", {
        CornerRadius = UDim.new(0, radius or 10)
    }, obj)
end

local function animate(obj, props, duration)
    TweenService:Create(
        obj,
        TweenInfo.new(
            duration or 0.2,
            Enum.EasingStyle.Quart,
            Enum.EasingDirection.Out
        ),
        props
    ):Play()
end

local function label(parent, text, height, color)
    return create("TextLabel", {
        Size = UDim2.new(1, 0, 0, height or 30),
        BackgroundTransparency = 1,
        Text = text,
        TextColor3 = color or C.Text,
        TextSize = 13,
        Font = Enum.Font.Gotham,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Top,
        TextWrapped = true
    }, parent)
end

-- isActive (optional): function returning true when the button
-- should keep the accent colour (toggles, selected tab, favourites).
local function button(parent, text, callback, isActive)
    local function baseColor()
        if isActive and isActive() then
            return C.Accent
        end
        return C.Button
    end

    local b = create("TextButton", {
        Size = UDim2.new(1, 0, 0, 38),
        BackgroundColor3 = baseColor(),
        Text = text,
        TextColor3 = C.Text,
        TextSize = 13,
        Font = Enum.Font.GothamMedium,
        AutoButtonColor = false
    }, parent)

    round(b, 9)

    b.MouseEnter:Connect(function()
        animate(b, {BackgroundColor3 = C.Accent})
    end)

    b.MouseLeave:Connect(function()
        animate(b, {BackgroundColor3 = baseColor()})
    end)

    b.Activated:Connect(function()
        if callback then
            callback()
        end
    end)

    return b
end

-- Switch row. Returns {Set = function(bool, silent), Get = function()}.
function UI.Toggle(parent, title, subtitle, default, callback)
    local state = default == true
    local row = rowFrame(parent, 50)
    local subtitleLabel = rowText(row, title, subtitle, 76)

    local track = create("Frame", {
        AnchorPoint = Vector2.new(1, 0.5),
        Position = UDim2.new(1, -12, 0.5, 0),
        Size = UDim2.fromOffset(44, 22),
        BackgroundColor3 = state and C.Accent or C.Panel,
        BorderSizePixel = 0
    }, row)

    round(track, 11)

    local knob = create("Frame", {
        Size = UDim2.fromOffset(18, 18),
        Position = state and UDim2.fromOffset(24, 2) or UDim2.fromOffset(2, 2),
        BackgroundColor3 = C.Text,
        BorderSizePixel = 0
    }, track)

    round(knob, 9)

    local click = create("TextButton", {
        Size = UDim2.new(1, 0, 1, 0),
        BackgroundTransparency = 1,
        Text = "",
        AutoButtonColor = false
    }, row)

    local object = {}

    local function paint()
        animate(track, {BackgroundColor3 = state and C.Accent or C.Panel}, 0.15)
        animate(knob, {Position = state and UDim2.fromOffset(24, 2) or UDim2.fromOffset(2, 2)}, 0.15)
    end

    function object.Set(value, silent)
        state = value == true
        paint()

        if not silent and callback then
            callback(state)
        end
    end

    function object.Get()
        return state
    end

    function object.SetSubtitle(text)
        subtitleLabel.Text = text
    end

    click.Activated:Connect(function()
        object.Set(not state)
    end)

    return object
end

-- Slider with mouse + touch dragging. Returns {Set = function(value, silent), Get = function()}.
function UI.Slider(parent, title, minValue, maxValue, step, default, format, callback)
    step = step or 1
    local value = math.clamp(default or minValue, minValue, maxValue)
    local row = rowFrame(parent, 58)

    create("TextLabel", {
        BackgroundTransparency = 1,
        Position = UDim2.fromOffset(12, 5),
        Size = UDim2.new(1, -110, 0, 20),
        Text = title,
        TextColor3 = C.Text,
        TextSize = 14,
        Font = Enum.Font.GothamMedium,
        TextXAlignment = Enum.TextXAlignment.Left
    }, row)

    local valueLabel = create("TextLabel", {
        BackgroundTransparency = 1,
        AnchorPoint = Vector2.new(1, 0),
        Position = UDim2.new(1, -12, 0, 5),
        Size = UDim2.fromOffset(90, 20),
        TextColor3 = C.Accent,
        TextSize = 13,
        Font = Enum.Font.GothamBold,
        TextXAlignment = Enum.TextXAlignment.Right
    }, row)

    local bar = create("Frame", {
        Position = UDim2.new(0, 12, 0, 36),
        Size = UDim2.new(1, -24, 0, 6),
        BackgroundColor3 = C.Panel,
        BorderSizePixel = 0
    }, row)

    round(bar, 3)

    local fill = create("Frame", {
        Size = UDim2.new(0, 0, 1, 0),
        BackgroundColor3 = C.Accent,
        BorderSizePixel = 0
    }, bar)

    round(fill, 3)

    local knob = create("Frame", {
        AnchorPoint = Vector2.new(0.5, 0.5),
        Size = UDim2.fromOffset(16, 16),
        Position = UDim2.new(0, 0, 0.5, 0),
        BackgroundColor3 = C.Text,
        BorderSizePixel = 0
    }, bar)

    round(knob, 8)

    local hit = create("TextButton", {
        Position = UDim2.new(0, 0, 0, 26),
        Size = UDim2.new(1, 0, 0, 30),
        BackgroundTransparency = 1,
        Text = "",
        AutoButtonColor = false
    }, row)

    local object = {}

    local function render()
        local alpha = 0

        if maxValue > minValue then
            alpha = (value - minValue) / (maxValue - minValue)
        end

        fill.Size = UDim2.new(alpha, 0, 1, 0)
        knob.Position = UDim2.new(alpha, 0, 0.5, 0)
        valueLabel.Text = format and format(value) or tostring(value)
    end

    function object.Set(newValue, silent)
        newValue = math.clamp(newValue, minValue, maxValue)
        newValue = math.floor((newValue - minValue) / step + 0.5) * step + minValue
        newValue = math.clamp(newValue, minValue, maxValue)

        local changed = newValue ~= value
        value = newValue
        render()

        if changed and not silent and callback then
            callback(value)
        end
    end

    function object.Get()
        return value
    end

    render()

    local scrollParent = parent:FindFirstAncestorOfClass("ScrollingFrame")
    local dragging = false

    local function updateFromX(x)
        local width = bar.AbsoluteSize.X
        if width <= 0 then return end

        local alpha = math.clamp((x - bar.AbsolutePosition.X) / width, 0, 1)
        object.Set(minValue + alpha * (maxValue - minValue))
    end

    local function stopDragging()
        if not dragging then return end

        dragging = false

        if scrollParent then
            scrollParent.ScrollingEnabled = true
        end
    end

    hit.InputBegan:Connect(function(inputObject)
        if inputObject.UserInputType == Enum.UserInputType.MouseButton1
            or inputObject.UserInputType == Enum.UserInputType.Touch then
            dragging = true

            if scrollParent then
                scrollParent.ScrollingEnabled = false
            end

            updateFromX(inputObject.Position.X)
        end
    end)

    track(UserInputService.InputChanged:Connect(function(inputObject)
        if dragging and (
            inputObject.UserInputType == Enum.UserInputType.MouseMovement
            or inputObject.UserInputType == Enum.UserInputType.Touch
        ) then
            updateFromX(inputObject.Position.X)
        end
    end))

    track(UserInputService.InputEnded:Connect(function(inputObject)
        if inputObject.UserInputType == Enum.UserInputType.MouseButton1
            or inputObject.UserInputType == Enum.UserInputType.Touch then
            stopDragging()
        end
    end))

    return object
end

-- Dropdown. options = {"A", "B", ...}. Returns {Set = function(index, silent), Get = function()}.
function UI.Dropdown(parent, title, options, defaultIndex, callback)
    local index = defaultIndex or 1

    local holder = create("Frame", {
        Size = UDim2.new(1, 0, 0, 0),
        AutomaticSize = Enum.AutomaticSize.Y,
        BackgroundTransparency = 1
    }, parent)

    create("UIListLayout", {
        Padding = UDim.new(0, 4),
        SortOrder = Enum.SortOrder.LayoutOrder
    }, holder)

    local head = create("TextButton", {
        Size = UDim2.new(1, 0, 0, 46),
        BackgroundColor3 = C.Button,
        Text = "",
        AutoButtonColor = false,
        LayoutOrder = 0
    }, holder)

    round(head, 9)

    local valueLabel = rowText(head, title, options[index], 50)

    local arrow = create("TextLabel", {
        BackgroundTransparency = 1,
        AnchorPoint = Vector2.new(1, 0.5),
        Position = UDim2.new(1, -12, 0.5, 0),
        Size = UDim2.fromOffset(20, 20),
        Text = ">",
        TextColor3 = C.Muted,
        TextSize = 14,
        Font = Enum.Font.GothamBold
    }, head)

    local list = create("Frame", {
        Size = UDim2.new(1, 0, 0, 0),
        AutomaticSize = Enum.AutomaticSize.Y,
        BackgroundTransparency = 1,
        LayoutOrder = 1,
        Visible = false
    }, holder)

    create("UIListLayout", {
        Padding = UDim.new(0, 4),
        SortOrder = Enum.SortOrder.LayoutOrder
    }, list)

    create("UIPadding", {PaddingLeft = UDim.new(0, 12)}, list)

    local object = {}
    local optionButtons = {}

    local function paint()
        valueLabel.Text = options[index]

        for i, optionButton in ipairs(optionButtons) do
            animate(optionButton, {BackgroundColor3 = i == index and C.Accent or C.Panel}, 0.12)
        end
    end

    function object.Set(newIndex, silent)
        if not options[newIndex] then return end

        index = newIndex
        paint()

        if not silent and callback then
            callback(index, options[index])
        end
    end

    function object.Get()
        return index, options[index]
    end

    for i, optionName in ipairs(options) do
        local optionButton = create("TextButton", {
            Size = UDim2.new(1, 0, 0, 32),
            BackgroundColor3 = C.Panel,
            Text = optionName,
            TextColor3 = C.Text,
            TextSize = 13,
            Font = Enum.Font.Gotham,
            AutoButtonColor = false,
            LayoutOrder = i
        }, list)

        round(optionButton, 8)
        optionButtons[i] = optionButton

        optionButton.Activated:Connect(function()
            object.Set(i)
            list.Visible = false
            arrow.Text = ">"
        end)
    end

    head.Activated:Connect(function()
        list.Visible = not list.Visible
        arrow.Text = list.Visible and "v" or ">"
    end)

    paint()
    return object
end

local EffectState = {}

local function toggleEffect(name, enabled)
    local fullName = EFFECT_PREFIX .. name

    local oldEffect = Lighting:FindFirstChild(fullName)
    if oldEffect then

local function requestAPI(method, path, body, extraHeaders)
    if not API_CONFIGURED then
        return false, "Адрес API не настроен (CONFIG.API_BASE)"
    end

    local requestFunction =
        (syn and syn.request)
        or (http and http.request)
        or http_request
        or request

    if not requestFunction then
        return false, "HTTP-запросы недоступны в этом executor"
    end

    local headers = {
        ["Content-Type"] = "application/json",
        ["X-DeltaHub-UserId"] = tostring(Player.UserId)
    }

    for k, v in pairs(extraHeaders or {}) do
        headers[k] = v
    end

    local encodedBody = nil

    if body then
        local encodedOK, encoded = pcall(function()
            return HttpService:JSONEncode(body)
        end)

        if not encodedOK then
            return false, "Не удалось подготовить запрос"
        end

        encodedBody = encoded
    end

    local success, response

    -- one automatic retry: network hiccups or a 5xx answer are common on the first call
    for attempt = 1, 2 do
        success, response = pcall(function()
            return requestFunction({
                Url = CONFIG.API_BASE .. path,
                Method = method,
                Headers = headers,
                Body = encodedBody
            })
        end)

        local status = 0

        if success and type(response) == "table" then
            status = response.StatusCode or 0
        end

        local retryable = (not success) or status == 0 or status >= 500

        if not retryable then
            break
        end

        if attempt == 1 then
            task.wait(0.8)
        end
    end

    if not success then
        return false, tostring(response)
    end

    if type(response) ~= "table" then
        return false, "Нет ответа от API"
    end

    local decodedOK, decoded = pcall(function()
        return HttpService:JSONDecode(response.Body)
    end)

    local status = response.StatusCode or 0

    if status < 200 or status >= 300 then
        local message = "HTTP " .. tostring(status)
        if decodedOK and type(decoded) == "table" and decoded.error then
            message = tostring(decoded.error)
        end
        return false, message
    end

    if not decodedOK then
        return false, "Некорректный ответ API"
    end

    return true, decoded
end

local function formatExpiry(ms)
    local ok, text = pcall(function()
        return os.date("!%d.%m.%Y %H:%M", math.floor(ms / 1000)) .. " UTC"
    end)

    return ok and text or tostring(ms)
end

local function doSearch()
    local query = (SearchInput.Text:gsub("^%s+", ""):gsub("%s+$", ""))

    if #query < 2 then
        SearchStatus.Text = "Введи минимум 2 символа ника."
        return
    end

    searchToken = searchToken + 1
    local token = searchToken

    SearchStatus.Text = "Ищем..."

    local found = {}
    local seen = {}

    local function push(userId, username, displayName)
        if type(userId) ~= "number" or type(username) ~= "string" then return end
        if seen[userId] or userId == Player.UserId then return end

        seen[userId] = true
        table.insert(found, {
            userId = userId,
            username = username,
            displayName = displayName
        })
    end

    -- 1) players on this server (instant)
    local lowered = query:lower()

    for _, p in ipairs(Players:GetPlayers()) do
        if p.Name:lower():find(lowered, 1, true)
            or p.DisplayName:lower():find(lowered, 1, true) then
            push(p.UserId, p.Name, p.DisplayName)
        end
    end

    renderResults(found)

    if not API_CONFIGURED then
        SearchStatus.Text =
            (#found > 0 and ("Найдено на сервере: " .. #found .. ". ") or "На сервере никого нет. ") ..
            "Поиск по всему Roblox недоступен: API не подключён."
        return
    end

    -- 2) the whole Roblox, through our server
    task.spawn(function()
        local ok, result = requestAPI(
            "GET",
            "/v1/search?q=" .. HttpService:UrlEncode(query)
        )

        if token ~= searchToken then return end

        if ok and type(result) == "table" then
            for _, user in ipairs(result.users or {}) do
                push(user.userId, user.username, user.displayName)
            end

            renderResults(found)

            SearchStatus.Text = #found > 0
                and ("Найдено: " .. #found)
                or "Никого не найдено."
        else
            SearchStatus.Text =
                (#found > 0 and ("Найдено на сервере: " .. #found .. ". ") or "") ..
                "Поиск по Roblox не удался: " .. tostring(result)
        end
    end)
end

local function refreshPremium(hint)
    setPremiumMessage("Проверяем Premium...", C.Accent, "")

    task.spawn(function()
        local key = LicenseKey

        if key == "" then
            applyPremiumState(nil)
            setPremiumMessage("Premium: ключ не введён", C.Muted, "Введи ключ, который выдал администратор")
            return
        end

        local headers = {["X-DeltaHub-Key"] = key}

        local ok, result = requestAPI(
            "GET",
            "/v1/premium/" .. tostring(Player.UserId),
            nil,
            headers
        )

        if not ok then
            setPremiumMessage("Premium: нет связи с сервисом", C.Muted, tostring(result))
            return
        end

        if not result.active then
            applyPremiumState(nil)
            setPremiumMessage(
                "Premium: НЕ АКТИВЕН",
                C.Muted,
                hint or "Ключ неверный, срок истёк или Premium не выдан"
            )
            return
        end

        local okFeatures, features = requestAPI("GET", "/v1/premium/features", nil, headers)

        if not okFeatures or type(features) ~= "table" or type(features.code) ~= "string" then
            setPremiumMessage("Premium: не удалось загрузить функции", C.Red, tostring(features))
            return
        end

        local loader, loadError = loadstring(features.code)

        if not loader then
            setPremiumMessage("Premium: ошибка загрузки", C.Red, tostring(loadError))
            return
        end

        local okRun, module = pcall(loader)

        if not okRun or type(module) ~= "table" or type(module.toggle) ~= "function" then
            setPremiumMessage("Premium: ошибка загрузки", C.Red, tostring(module))
            return
        end

        applyPremiumState(module)

        setPremiumMessage(
            "Premium: АКТИВЕН",
            C.Green,
            result.expiresAt
                and ("Действует до: " .. formatExpiry(result.expiresAt))
                or "Срок: навсегда"
        )
    end)
end

    end)
end)

    -- put it in your own launcher line (not in this script):
    --   getgenv().DeltaHubAdminKey = "your-secret"
    local function adminHeaders()
        local headers = {}

        local ok, secret = pcall(function()
            return getgenv and getgenv().DeltaHubAdminKey
        end)

        if ok and type(secret) == "string" and secret ~= "" then
            headers["X-DeltaHub-AdminKey"] = secret
        end

        return headers
    end

    local function adminCall(method, path, body, onSuccess)
        AdminOutput.Text = "Запрос..."
        AdminOutput.TextColor3 = C.Muted

        task.spawn(function()
            local ok, result = requestAPI(method, path, body, adminHeaders())

            if ok then
                onSuccess(result)
            else
                showResult(false, "Ошибка: " .. tostring(result))
            end
        end)
    end

    -- Duration: the "days" field wins if filled, otherwise the button above.
    -- Returns "<N>d" / "permanent" and a readable label, or nil on bad input.
    local function readDuration()
        local daysText = (DaysInput.Text:gsub("%s+", ""))

        if daysText == "" then
            local option = DurationOptions[durationIndex]
            return option[2], option[1]
        end

        local days = tonumber(daysText)

        if not days or days < 0 or days > 3650 or days % 1 ~= 0 then
            showResult(false, "Срок в днях: целое число от 0 до 3650 (0 = навсегда).")
            return nil
        end

        if days == 0 then
            return "permanent", "навсегда"
        end

        return days .. "d", days .. " дн."
    end

    local function readTarget()
        local target = (TargetInput.Text:gsub("^%s+", ""):gsub("%s+$", ""))

        if target == "" then
            showResult(false, "Введи ник или Roblox ID игрока.")
            return nil
        end

        return target
    end

    local function premiumAction(action)
        local target = readTarget()
        if not target then return end

        local duration = "1d"

        if action == "grant" then
            duration = readDuration()
            if not duration then return end
        end

        adminCall(
            "POST",
            "/v1/admin/premium",
            {action = action, target = target, duration = duration},
            function(result)
                local text = result.message or "Готово"

                if action == "grant" then
                    if result.expiresAt then
                        text = text .. "\nДо: " .. formatExpiry(result.expiresAt)
                    else
                        text = text .. "\nСрок: навсегда"
                    end
                elseif action == "newkey" and result.expiresAt then
                    text = text .. "\nДо: " .. formatExpiry(result.expiresAt)
                end

                if result.key then
                    text = text .. "\nКлюч игрока: " .. result.key
                end

                showResult(true, text)
            end
        )
    end

local TabButtons = {}

local currentTab

local function openTab(name)
    currentTab = name
    PageTitle.Text = name

    for pageName, page in pairs(Pages) do
        page.Visible = pageName == name
    end

    for tabName, tabButton in pairs(TabButtons) do
        animate(tabButton, {
            BackgroundColor3 = tabName == name and C.Accent or C.Button
        })

        if TabBars[tabName] then
            TabBars[tabName].Visible = tabName == name
        end
    end
end

local sidebarOrder = 0

for _, group in ipairs(tabGroups) do
    local available = {}

    for _, name in ipairs(group[2]) do
        -- the admin page exists only for the owner (see IsAdmin)
        if Pages[name] then
            table.insert(available, name)
        end
    end

    if #available > 0 then
        sidebarOrder = sidebarOrder + 1

        create("TextLabel", {
            Size = UDim2.new(1, 0, 0, 16),
            BackgroundTransparency = 1,
            Text = group[1],
            TextColor3 = C.Muted,
            TextSize = 10,
            Font = Enum.Font.GothamBold,
            TextXAlignment = Enum.TextXAlignment.Left,
            LayoutOrder = sidebarOrder
        }, Sidebar)

        for _, name in ipairs(available) do
            sidebarOrder = sidebarOrder + 1

            local tabButton = button(
                Sidebar,
                name,
                function()
                    openTab(name)
                end,
                function()
                    return currentTab == name
                end
            )

            tabButton.Size = UDim2.new(1, 0, 0, 32)
            tabButton.LayoutOrder = sidebarOrder
            tabButton.TextXAlignment = Enum.TextXAlignment.Left

                Size = UDim2.new(0, 3, 0, 16),

                BackgroundColor3 = C.Text,
                BorderSizePixel = 0,
                Visible = false
            }, tabButton)

            round(bar, 2)
            TabBars[name] = bar
        end
    end
end

-- DRAGGING (mouse and touch)

local dragStart
local startPosition

track(Top.InputBegan:Connect(function(inputObject)
    if inputObject.UserInputType == Enum.UserInputType.MouseButton1
        or inputObject.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = inputObject.Position
        startPosition = Main.Position
    end
end))

track(UserInputService.InputChanged:Connect(function(inputObject)
    if dragging and (
        inputObject.UserInputType == Enum.UserInputType.MouseMovement
        or inputObject.UserInputType == Enum.UserInputType.Touch
    ) then
        local delta = inputObject.Position - dragStart

        Main.Position = UDim2.new(
            startPosition.X.Scale, startPosition.X.Offset + delta.X,
            startPosition.Y.Scale, startPosition.Y.Offset + delta.Y
        )
    end
end))

track(UserInputService.InputEnded:Connect(function(inputObject)
    if inputObject.UserInputType == Enum.UserInputType.MouseButton1
        or inputObject.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end))
